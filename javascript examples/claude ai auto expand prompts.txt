Longer Version - automation using tempermonkey extension in brave browser:

// ==UserScript==
// @name         Claude Auto-Expand Show More
// @namespace    http://tampermonkey.net/
// @version      1.9
// @description  Automatically expand all "Show more" buttons and tool-use blocks in Claude
// @match        *://claude.ai/*
// @match        *://*.claude.ai/*
// @grant        none
// @run-at       document-start
// ==/UserScript==
(function() {
    'use strict';
    console.log('Claude Auto-Expand: Script loaded at document-start');

    function expandAll() {
        let count = 0;

        // 1. "Show more" text buttons (original behaviour)
        document.querySelectorAll('button').forEach(btn => {
            if (btn.textContent.trim().toLowerCase() === 'show more') {
                btn.click();
                count++;
            }
        });

        // 2. ALL tool-use blocks universally (fetch, view files, skills, search, etc.)
        //    Strategy: find any tool-status-caret, walk up to its collapsed button parent.
        //    tool-status-caret ONLY appears inside tool blocks, never on mic/nav/other UI.
        document.querySelectorAll('[data-testid="tool-status-caret"]').forEach(caret => {
            const btn = caret.closest('button[aria-expanded="false"]');
            if (btn) {
                btn.click();
                count++;
            }
        });

        if (count > 0) {
            console.log(`Claude Auto-Expand: Clicked ${count} button(s)`);
        }
    }

    setTimeout(() => { console.log('Claude Auto-Expand: First attempt (2s)');  expandAll(); }, 2000);
    setTimeout(() => { console.log('Claude Auto-Expand: Second attempt (5s)'); expandAll(); }, 5000);
    setTimeout(() => {
        console.log('Claude Auto-Expand: Third attempt (10s)');
        expandAll();

        let debounceTimer;
        const observer = new MutationObserver(() => {
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(expandAll, 300);
        });

        if (document.body) {
            observer.observe(document.body, { childList: true, subtree: true });
            console.log('Claude Auto-Expand: MutationObserver started');
        }
    }, 10000);
})();
